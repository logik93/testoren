cmake_minimum_required(VERSION 3.22)
project(Logisth VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(LOGISTH_PLUGINS "Build JUCE plug-ins" ON)
option(LOGISTH_VST2 "Build VST2 using your licensed SDK" OFF)
set(VST2_SDK_PATH "" CACHE PATH "Path to licensed VST2 SDK")
if(BUILD_TESTING)
  add_executable(engine_tests tests/EngineTests.cpp)
  target_include_directories(engine_tests PRIVATE Source)
  add_test(NAME engine COMMAND engine_tests)
  foreach(suite IN ITEMS Modulation Arpeggiator Hybrid Studio)
    add_executable(${suite}_tests tests/${suite}Tests.cpp)
    target_include_directories(${suite}_tests PRIVATE Source)
    add_test(NAME ${suite} COMMAND ${suite}_tests)
  endforeach()
endif()
if(NOT LOGISTH_PLUGINS)
  return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(JUCE)
set(formats VST3 Standalone)
if(LOGISTH_VST2)
  if(NOT EXISTS "${VST2_SDK_PATH}/pluginterfaces/vst2.x/aeffect.h")
    message(FATAL_ERROR "VST2 needs your licensed SDK with pluginterfaces/vst2.x/aeffect.h. See README.md.")
  endif()
  juce_set_vst2_sdk_path("${VST2_SDK_PATH}")
  list(APPEND formats VST)
endif()
juce_add_plugin(Logisth COMPANY_NAME "Testoren" BUNDLE_ID "com.testoren.logisth"
  PLUGIN_MANUFACTURER_CODE Tstr PLUGIN_CODE Lgst
  FORMATS ${formats} PRODUCT_NAME "logisth"
  IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE
  IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Synth)
juce_add_binary_data(LogisthFonts SOURCES
  assets/fonts/OpenSans-Regular.ttf assets/fonts/OpenSans-Semibold.ttf)
target_sources(Logisth PRIVATE Source/PluginProcessor.cpp Source/SourceAssets.cpp Source/PresetLibrary.cpp Source/PluginEditor.cpp)
target_link_libraries(Logisth PRIVATE LogisthFonts)
target_compile_definitions(Logisth PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Logisth PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
FetchContent_Declare(clap_juce_extensions GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git GIT_TAG 6bde108329ff9fc24404b79b8b397261a75cf6a8)
FetchContent_MakeAvailable(clap_juce_extensions)
clap_juce_extensions_plugin(TARGET Logisth CLAP_ID "com.testoren.logisth" CLAP_FEATURES instrument synthesizer stereo)
if(BUILD_TESTING)
  add_executable(processor_tests tests/ProcessorTests.cpp)
  target_include_directories(processor_tests PRIVATE Source)
  target_link_libraries(processor_tests PRIVATE Logisth juce::juce_audio_utils juce::juce_dsp)
  add_test(NAME processor COMMAND processor_tests)
endif()

if(BUILD_TESTING AND UNIX AND NOT APPLE)
  add_executable(clap_smoke tests/ClapSmoke.cpp)
  target_include_directories(clap_smoke PRIVATE Source  "${clap_juce_extensions_SOURCE_DIR}/clap-libs/clap/include")
  target_link_libraries(clap_smoke PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME clap_binary COMMAND clap_smoke $<TARGET_FILE:Logisth_CLAP>)
endif()

if(BUILD_TESTING AND UNIX AND NOT APPLE)
  add_executable(vst3_smoke tests/Vst3Smoke.cpp)
  target_include_directories(vst3_smoke PRIVATE Source  "${juce_SOURCE_DIR}/modules/juce_audio_processors/format_types/VST3_SDK")
  target_link_libraries(vst3_smoke PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME vst3_binary COMMAND vst3_smoke $<TARGET_FILE:Logisth_VST3>)
endif()
